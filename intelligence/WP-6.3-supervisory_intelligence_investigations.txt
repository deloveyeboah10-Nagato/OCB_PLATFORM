/*
====================================================================================================================================
    1. SikaCredit: Lending surge and debt recycling

    Question:
    > During SikaCredit's November 2025 lending surge, which customers exhibit elevated debt-recycling repayment-deficiency,
      reborrowing-rapidity and recurrence scores?

    Institutional trigger:
    > November principal disbursements increased from GHS 104,524 in October to GHS 210,646.56. The number of loans issued rose
      from 82 to 146.
====================================================================================================================================
*/
-- 1. Institutional trigger: November lending surge
WITH monthly_disbursements
AS (
    SELECT
        CONVERT(CHAR(7), disbursement_timestamp, 120) AS year_month,
        COUNT(DISTINCT ocb_customer_id)               AS distinct_loan_customers,
        SUM(principal_amount)                         AS principal_disbursed,
        COUNT(*)                                      AS principal_disbursed_count,
        AVG(principal_amount)                         AS average_principal_disbursed
    FROM gold.vw_loan
    GROUP BY CONVERT(CHAR(7), disbursement_timestamp, 120)
    ),
monthly_repayments
AS (
    SELECT
        CONVERT(CHAR(7), repayment_timestamp, 120) AS year_month,
        COUNT(DISTINCT ocb_customer_id)            AS distinct_repayment_customers,
        SUM(repayment_amount)                      AS principal_repaid,
        COUNT(*)                                   AS principal_repaid_count,
        AVG(repayment_amount)                      AS average_principal_repaid
    FROM gold.vw_repayment
    GROUP BY CONVERT(CHAR(7), repayment_timestamp, 120)
    ),
monthly_flows
AS (
    SELECT
        COALESCE(d.year_month, r.year_month)        AS year_month,
        COALESCE(d.distinct_loan_customers, 0)      AS distinct_loan_customers,
        COALESCE(d.principal_disbursed, 0)          AS principal_disbursed,
        COALESCE(d.principal_disbursed_count, 0)    AS principal_disbursed_count,
        COALESCE(d.average_principal_disbursed, 0)  AS average_principal_disbursed,
        COALESCE(r.distinct_repayment_customers, 0) AS distinct_repayment_customers,
        COALESCE(r.principal_repaid, 0)             AS principal_repaid,
        COALESCE(r.principal_repaid_count, 0)       AS principal_repaid_count,
        COALESCE(r.average_principal_repaid, 0)     AS average_principal_repaid
    FROM monthly_disbursements AS d
    FULL OUTER JOIN monthly_repayments AS r ON
            d.year_month = r.year_month
    ),
monthly_changes
AS (
    SELECT
        *,
        LAG(distinct_loan_customers) OVER (
            ORDER BY year_month
            ) AS prev_loan_customers,
        LAG(principal_disbursed) OVER (
            ORDER BY year_month
            ) AS prev_principal_disbursed,
        LAG(principal_disbursed_count) OVER (
            ORDER BY year_month
            ) AS prev_disbursement_count,
        LAG(average_principal_disbursed) OVER (
            ORDER BY year_month
            ) AS prev_average_disbursed,
        LAG(distinct_repayment_customers) OVER (
            ORDER BY year_month
            ) AS prev_repayment_customers,
        LAG(principal_repaid) OVER (
            ORDER BY year_month
            ) AS prev_principal_repaid,
        LAG(principal_repaid_count) OVER (
            ORDER BY year_month
            ) AS prev_repayment_count,
        LAG(average_principal_repaid) OVER (
            ORDER BY year_month
            ) AS prev_average_repaid
    FROM monthly_flows
    )
SELECT
    year_month,
    -- Disbursement metrics: November values and changes from October
    distinct_loan_customers,
    distinct_loan_customers - prev_loan_customers                                                                                   AS loan_customers_change,
    CAST(100.0 * (distinct_loan_customers - prev_loan_customers) / NULLIF(prev_loan_customers, 0) AS DECIMAL(10, 2))                AS loan_customers_change_pct,
    principal_disbursed_count,
    principal_disbursed_count - prev_disbursement_count                                                                             AS disbursement_count_change,
    CAST(100.0 * (principal_disbursed_count - prev_disbursement_count) / NULLIF(prev_disbursement_count, 0) AS DECIMAL(10, 2))      AS disbursement_count_change_pct,
    CAST(principal_disbursed AS DECIMAL(19, 3))                                                                                     AS principal_disbursed,
    CAST(principal_disbursed - prev_principal_disbursed AS DECIMAL(19, 3))                                                          AS disbursement_sum_change,
    CAST(100.0 * (principal_disbursed - prev_principal_disbursed) / NULLIF(prev_principal_disbursed, 0) AS DECIMAL(10, 2))          AS disbursement_sum_change_pct,
    CAST(average_principal_disbursed AS DECIMAL(19, 3))                                                                             AS average_principal_disbursed,
    CAST(average_principal_disbursed - prev_average_disbursed AS DECIMAL(19, 3))                                                    AS average_disbursement_change,
    CAST(100.0 * (average_principal_disbursed - prev_average_disbursed) / NULLIF(prev_average_disbursed, 0) AS DECIMAL(10, 2))      AS average_disbursement_change_pct,
    -- Repayment metrics: November values and changes from October
    distinct_repayment_customers,
    distinct_repayment_customers - prev_repayment_customers                                                                         AS repayment_customers_change,
    CAST(100.0 * (distinct_repayment_customers - prev_repayment_customers) / NULLIF(prev_repayment_customers, 0) AS DECIMAL(10, 2)) AS repayment_customers_change_pct,
    principal_repaid_count,
    principal_repaid_count - prev_repayment_count                                                                                   AS repayment_count_change,
    CAST(100.0 * (principal_repaid_count - prev_repayment_count) / NULLIF(prev_repayment_count, 0) AS DECIMAL(10, 2))               AS repayment_count_change_pct,
    CAST(principal_repaid AS DECIMAL(19, 3))                                                                                        AS principal_repaid,
    CAST(principal_repaid - prev_principal_repaid AS DECIMAL(19, 3))                                                                AS repayment_sum_change,
    CAST(100.0 * (principal_repaid - prev_principal_repaid) / NULLIF(prev_principal_repaid, 0) AS DECIMAL(10, 2))                   AS repayment_sum_change_pct,
    CAST(average_principal_repaid AS DECIMAL(19, 3))                                                                                AS average_principal_repaid,
    CAST(average_principal_repaid - prev_average_repaid AS DECIMAL(19, 3))                                                          AS average_repayment_change,
    CAST(100.0 * (average_principal_repaid - prev_average_repaid) / NULLIF(prev_average_repaid, 0) AS DECIMAL(10, 2))               AS average_repayment_change_pct
FROM monthly_changes
WHERE year_month >= '2025-10'
        AND year_month < '2025-12';
GO

-- 2. SikaCredit Debt Recycling: Customer-Level Signal Aggregation for November Only (flag)
CREATE
        OR

ALTER VIEW intelligence.vw_sikacredit_debt_recycling_customer_signals
AS
WITH customer_deficiency
AS (
    SELECT
        ocb_customer_id,
        CONVERT(CHAR(7), next_loan_timestamp, 120) AS year_month,
        1                                          AS repayment_deficiency_flag
    FROM intelligence.vw_repayment_deficiency_score
    WHERE repayment_deficiency_mad_score > 0
    GROUP BY ocb_customer_id,
        CONVERT(CHAR(7), next_loan_timestamp, 120)
    ),
customer_rapidity
AS (
    SELECT
        ocb_customer_id,
        CONVERT(CHAR(7), next_loan_timestamp, 120) AS year_month,
        1                                          AS reborrowing_rapidity_flag
    FROM intelligence.vw_reborrowing_rapidity_score
    WHERE reborrowing_rapidity_mad_score > 0
    GROUP BY ocb_customer_id,
        CONVERT(CHAR(7), next_loan_timestamp, 120)
    ),
customer_recurrence
AS (
    SELECT
        ocb_customer_id,
        CONVERT(CHAR(7), next_recycling_event_timestamp, 120) AS year_month,
        1                                                     AS recycling_recurrence_flag
    FROM intelligence.vw_debt_recycling_recurrence_score
    WHERE debt_recycling_recurrence_mad_score > 0
    GROUP BY ocb_customer_id,
        CONVERT(CHAR(7), next_recycling_event_timestamp, 120)
    ),
customer_signals
AS (
    SELECT
        COALESCE(d.ocb_customer_id, r.ocb_customer_id, c.ocb_customer_id) AS ocb_customer_id,
        COALESCE(d.year_month, r.year_month, c.year_month)                AS year_month,
        COALESCE(d.repayment_deficiency_flag, 0)                          AS repayment_deficiency_flag,
        COALESCE(r.reborrowing_rapidity_flag, 0)                          AS reborrowing_rapidity_flag,
        COALESCE(c.recycling_recurrence_flag, 0)                          AS recycling_recurrence_flag
    FROM customer_deficiency AS d
    FULL OUTER JOIN customer_rapidity AS r ON
            r.ocb_customer_id = d.ocb_customer_id
                AND r.year_month = d.year_month
    FULL OUTER JOIN customer_recurrence AS c ON
            c.ocb_customer_id = COALESCE(d.ocb_customer_id, r.ocb_customer_id)
                AND c.year_month = COALESCE(d.year_month, r.year_month)
    ),
customer_classification
AS (
    SELECT
        ocb_customer_id,
        year_month,
        repayment_deficiency_flag,
        reborrowing_rapidity_flag,
        recycling_recurrence_flag,
        repayment_deficiency_flag + reborrowing_rapidity_flag + recycling_recurrence_flag AS signal_count
    FROM customer_signals
    )
SELECT
    ocb_customer_id,
    year_month,
    repayment_deficiency_flag,
    reborrowing_rapidity_flag,
    recycling_recurrence_flag,
    signal_count,
    CASE signal_count
        WHEN 1
            THEN '1 Signal'
        WHEN 2
            THEN '2 Signals'
        WHEN 3
            THEN 'All 3 Signals'
        ELSE 'No Signals'
        END AS signal_classification
FROM customer_classification;
GO

-- Which Customers Exhibited Elevated Debt-recycling signals, during the November, 2025 surge?
SELECT
    *
FROM intelligence.vw_sikacredit_debt_recycling_customer_signals
WHERE year_month = '2025-11'
ORDER BY signal_count DESC,
    ocb_customer_id;
GO

/*
====================================================================================================================================
    2. SikaCredit: What kind of borrowers drove the surge?

    Question:
    > Was SikaCredit's November 2025 lending surge driven primarily by first-time borrowers, repeat borrowers, or customers
      exhibiting unusually frequent borrowing and short inter-loan intervals?

    > Institutional trigger:
      The number of loans issued nearly doubled in November, contributing to a sharp increase in principal disbursements.
====================================================================================================================================
*/
-- Borrower composition
WITH customer_borrowing_history
AS (
    SELECT
        ocb_customer_id,
        MIN(disbursement_timestamp) AS first_loan_timestamp
    FROM gold.vw_loan
    GROUP BY ocb_customer_id
    ),
november_borrowers
AS (
    SELECT
        l.ocb_customer_id,
        l.loan_id,
        l.disbursement_timestamp,
        l.principal_amount,
        CASE 
            WHEN h.first_loan_timestamp >= '2025-11-01'
                    AND h.first_loan_timestamp < '2025-12-01'
                THEN 'First-Time Borrower'
            ELSE 'Repeat Borrower'
            END AS borrower_type
    FROM gold.vw_loan AS l
    INNER JOIN customer_borrowing_history AS h ON
            h.ocb_customer_id = l.ocb_customer_id
    WHERE l.disbursement_timestamp >= '2025-11-01'
            AND l.disbursement_timestamp < '2025-12-01'
    )
SELECT
    borrower_type,
    COUNT(DISTINCT ocb_customer_id) AS distinct_customers,
    COUNT(*)                        AS loans_issued,
    SUM(principal_amount)           AS principal_disbursed,
    AVG(principal_amount)           AS average_loan_principal
FROM november_borrowers
GROUP BY borrower_type
ORDER BY borrower_type;
GO

-- Borrowing behaviour of group
CREATE
        OR

ALTER VIEW intelligence.vw_sikacredit_borrowing_surge
AS
WITH customer_borrowing_history
AS (
    SELECT
        ocb_customer_id,
        MIN(disbursement_timestamp) AS first_loan_timestamp
    FROM gold.vw_loan
    GROUP BY ocb_customer_id
    ),
customer_monthly_borrowing
AS (
    SELECT
        ocb_customer_id,
        loan_id,
        disbursement_timestamp,
        principal_amount,
        COUNT(*) OVER (
            PARTITION BY ocb_customer_id,
            YEAR(disbursement_timestamp),
            MONTH(disbursement_timestamp)
            ) AS monthly_borrowing_frequency
    FROM gold.vw_loan
    ),
borrower_loans
AS (
    SELECT
        l.ocb_customer_id,
        l.loan_id,
        l.disbursement_timestamp,
        l.principal_amount,
        l.monthly_borrowing_frequency,
        CASE 
            WHEN h.first_loan_timestamp = l.disbursement_timestamp
                THEN 'First-Time Borrower'
            ELSE 'Repeat Borrower'
            END AS borrower_type
    FROM customer_monthly_borrowing AS l
    INNER JOIN customer_borrowing_history AS h ON
            h.ocb_customer_id = l.ocb_customer_id
    ),
reborrowing
AS (
    SELECT
        r.ocb_customer_id,
        r.next_loan_id,
        r.next_loan_timestamp,
        r.reborrowing_delay_minutes,
        r.reborrowing_rapidity_mad_score
    FROM intelligence.vw_reborrowing_rapidity_score AS r
    ),
recurrence
AS (
    SELECT
        c.ocb_customer_id,
        c.next_recycling_event_id,
        c.next_recycling_event_timestamp,
        c.recycling_recurrence_gap_minutes,
        c.debt_recycling_recurrence_mad_score
    FROM intelligence.vw_debt_recycling_recurrence_score AS c
    )
SELECT
    l.ocb_customer_id,
    l.loan_id,
    l.disbursement_timestamp,
    l.principal_amount,
    l.borrower_type,
    l.monthly_borrowing_frequency,
    r.reborrowing_delay_minutes,
    r.reborrowing_rapidity_mad_score,
    c.recycling_recurrence_gap_minutes,
    c.debt_recycling_recurrence_mad_score
FROM borrower_loans AS l
LEFT JOIN reborrowing AS r ON
        r.next_loan_id = l.loan_id
LEFT JOIN recurrence AS c ON
        c.next_recycling_event_id = l.loan_id;
GO

-- What kind of borrowers drove the November, 2025 surge?
SELECT
    ocb_customer_id,
    principal_amount,
    borrower_type,
    monthly_borrowing_frequency,
    reborrowing_rapidity_mad_score,
    debt_recycling_recurrence_mad_score
FROM intelligence.vw_sikacredit_borrowing_surge
WHERE disbursement_timestamp >= '2025-11-01'
        AND disbursement_timestamp < '2025-12-01'
ORDER BY ocb_customer_id,
    disbursement_timestamp;
GO

/*
====================================================================================================================================
    3. SikaCredit: Credit escalation during the lending surge

    Question:
    > Among customers receiving loans during the November 2025 lending surge, which exhibit elevated credit-value escalation,
      escalation recurrence or credit-obligation escalation signals?

    Institutional trigger:
    > November's sharp increase in lending raises a separate question about the size and progression of individual borrowers' credit
      exposure.
====================================================================================================================================
*/
GO

CREATE
        OR

ALTER VIEW intelligence.vw_sikacredit_credit_escalation_customer_signals
AS
WITH customer_escalation_score
AS (
    SELECT
        ocb_customer_id,
        loan_id,
        loan_date,
        1 AS escalation_value_flag
    FROM intelligence.vw_credit_value_escalation_score
    WHERE loan_escalation_mad_score > 0
    GROUP BY ocb_customer_id,
        loan_date,
        loan_id
    ),
customer_recurrence
AS (
    SELECT
        ocb_customer_id,
        loan_id,
        loan_date,
        1 AS escalation_recurrence_flag
    FROM intelligence.vw_credit_escalation_recurrence_score
    WHERE escalation_recurrence_mad_score > 0
    GROUP BY ocb_customer_id,
        loan_date,
        loan_id
    ),
customer_obligation
AS (
    SELECT
        ocb_customer_id,
        DATEFROMPARTS(YEAR(disbursement_timestamp), MONTH(disbursement_timestamp), 1) AS loan_date,
        loan_id,
        strike_1_pre_maturity_reborrowing,
        strike_2_maturity_breach,
        strike_3_post_maturity_reborrowing
    FROM intelligence.vw_credit_obligation_escalation_flag
    WHERE strike_1_pre_maturity_reborrowing IS NOT NULL
            AND strike_2_maturity_breach IS NOT NULL
            AND strike_3_post_maturity_reborrowing IS NOT NULL
    GROUP BY ocb_customer_id,
        DATEFROMPARTS(YEAR(disbursement_timestamp), MONTH(disbursement_timestamp), 1),
        loan_id,
        strike_1_pre_maturity_reborrowing,
        strike_2_maturity_breach,
        strike_3_post_maturity_reborrowing
    ),
customer_signals
AS (
    SELECT
        COALESCE(co.ocb_customer_id, es.ocb_customer_id, r.ocb_customer_id) AS ocb_customer_id,
        COALESCE(co.loan_date, r.loan_date, es.loan_date)                   AS year_month,
        COALESCE(es.escalation_value_flag, 0)                               AS escalation_value_flag,
        COALESCE(r.escalation_recurrence_flag, 0)                           AS escalation_recurrence_flag,
        COALESCE(co.strike_1_pre_maturity_reborrowing, 0)                   AS strike_1_pre_maturity_reborrowing,
        COALESCE(co.strike_2_maturity_breach, 0)                            AS strike_2_maturity_breach,
        COALESCE(co.strike_3_post_maturity_reborrowing, 0)                  AS strike_3_post_maturity_reborrowing
    FROM customer_obligation co
    FULL OUTER JOIN customer_recurrence AS r ON
            r.ocb_customer_id = co.ocb_customer_id
                AND r.loan_id = co.loan_id
                AND r.loan_date = co.loan_date
    FULL OUTER JOIN customer_escalation_score AS es ON
            es.ocb_customer_id = COALESCE(co.ocb_customer_id, r.ocb_customer_id)
                AND es.loan_id = COALESCE(co.loan_id, r.loan_id)
                AND es.loan_date = COALESCE(co.loan_date, r.loan_date)
    ),
signal_classification
AS (
    SELECT
        *,
        escalation_value_flag + escalation_recurrence_flag + strike_1_pre_maturity_reborrowing + strike_2_maturity_breach + strike_3_post_maturity_reborrowing AS signal_count
    FROM customer_signals
    )
SELECT
    *,
    CASE 
        WHEN signal_count = 1
            THEN '1 Signal'
        WHEN signal_count = 2
            THEN '2 Signals'
        WHEN signal_count >= 3
            THEN '3+ Signals'
        ELSE 'No Signals'
        END AS signal_classification
FROM signal_classification;
GO

-- Which Customers Exhibited Loan Escalation Signals For The November,2025 Surge?
SELECT
    *
FROM intelligence.vw_sikacredit_credit_escalation_customer_signals
WHERE year_month = '2025-11-01'
ORDER BY signal_count DESC,
    ocb_customer_id;
GO

-- Customer Transaction Analysis
CREATE
        OR

ALTER VIEW intelligence.vw_ananse_customer_transaction_analysis
AS
WITH monthly_totals
AS (
    SELECT
        CONVERT(CHAR(7), transaction_timestamp, 120) AS year_month,
        ocb_customer_id,
        COUNT(*)                                     AS total_transaction_count,
        SUM(transaction_amount)                      AS total_transaction_value,
        AVG(transaction_amount)                      AS average_transaction_value
    FROM gold.vw_transaction
    WHERE transaction_status = 'successful'
            AND transaction_type <> 'merchant payment'
    GROUP BY CONVERT(CHAR(7), transaction_timestamp, 120),
        ocb_customer_id
    ),
monthly_medians
AS (
    SELECT DISTINCT
        CONVERT(CHAR(7), transaction_timestamp, 120) AS year_month,
        ocb_customer_id,
        PERCENTILE_CONT(0.50) WITHIN
    GROUP (
            ORDER BY transaction_amount
            ) OVER (PARTITION BY CONVERT(CHAR(7), transaction_timestamp, 120)) AS median_transaction_value
    FROM gold.vw_transaction
    WHERE transaction_status = 'successful'
            AND transaction_type <> 'merchant payment'
    ),
monthly_summary
AS (
    SELECT
        t.*,
        m.median_transaction_value
    FROM monthly_totals t
    INNER JOIN monthly_medians m ON
            t.year_month = m.year_month
                AND t.ocb_customer_id = m.ocb_customer_id
    ),
monthly_changes
AS (
    SELECT
        *,
        LAG(total_transaction_count) OVER (
            PARTITION BY ocb_customer_id ORDER BY year_month
            ) AS previous_transaction_count,
        LAG(total_transaction_value) OVER (
            PARTITION BY ocb_customer_id ORDER BY year_month
            ) AS previous_transaction_value
    FROM monthly_summary
    )
SELECT
    year_month,
    ocb_customer_id,
    previous_transaction_count,
    total_transaction_count,
    total_transaction_count - previous_transaction_count                                                                                     AS transaction_count_mom_change,
    CAST(ROUND((total_transaction_count - previous_transaction_count) * 100.0 / NULLIF(previous_transaction_count, 0), 3) AS DECIMAL(19, 3)) AS transaction_count_mom_change_pct,
    CAST(ROUND(previous_transaction_value, 3) AS DECIMAL(19, 3))                                                                             AS previous_transaction_value,
    CAST(ROUND(total_transaction_value, 3) AS DECIMAL(19, 3))                                                                                AS total_transaction_value,
    CAST(ROUND((total_transaction_value - previous_transaction_value), 3) AS DECIMAL(19, 3))                                                 AS transaction_value_mom_change,
    CAST(ROUND((total_transaction_value - previous_transaction_value) * 100.0 / NULLIF(previous_transaction_value, 0), 3) AS DECIMAL(19, 3)) AS transaction_value_mom_change_pct,
    CAST(ROUND(average_transaction_value, 3) AS DECIMAL(19, 3))                                                                              AS average_transaction_value,
    CAST(ROUND(median_transaction_value, 3) AS DECIMAL(19, 3))                                                                               AS median_transaction_value
FROM monthly_changes;
GO

-- Which Customers Contributed Materially to Ananse's December 2025 Transaction Surge?
SELECT TOP (50)
    ocb_customer_id,
    year_month,
    previous_transaction_count   AS november_transaction_count,
    total_transaction_count      AS december_transaction_count,
    transaction_count_mom_change AS transaction_count_change,
    previous_transaction_value   AS november_transaction_value,
    total_transaction_value      AS december_transaction_value,
    transaction_value_mom_change AS transaction_value_change,
    transaction_count_mom_change_pct,
    transaction_value_mom_change_pct
FROM intelligence.vw_ananse_customer_transaction_analysis
WHERE year_month = '2025-12'
ORDER BY transaction_value_mom_change DESC;
GO

-- What TAE Signals Did The Top 50 Customers Who Contributed Materially to Ananse's December 2025 Transaction Surge Exhibit?
WITH transaction_level
AS (
    SELECT
        t.transaction_id,
        t.ocb_customer_id,
        CONVERT(CHAR(7), t.transaction_timestamp, 120) AS year_month,
        t.transaction_timestamp,
        t.transaction_type,
        t.transaction_amount,
        -- Global Amount
        g.log_transaction_amount,
        g.global_amount_score,
        -- Velocity
        v.txn_interval_mins,
        v.velocity_score
    FROM gold.vw_transaction AS t
    LEFT JOIN intelligence.vw_signal_global_score AS g ON
            t.transaction_id = g.transaction_id
                AND t.ocb_customer_id = g.ocb_customer_id
    LEFT JOIN intelligence.vw_signal_velocity_score AS v ON
            t.transaction_id = v.transaction_id
                AND t.ocb_customer_id = v.ocb_customer_id
    WHERE t.transaction_status = 'successful'
            AND t.transaction_type <> 'merchant payment'
    ),
top_50
AS (
    SELECT TOP (50)
        ocb_customer_id,
        year_month,
        previous_transaction_count   AS november_transaction_count,
        total_transaction_count      AS december_transaction_count,
        transaction_count_mom_change AS transaction_count_change,
        previous_transaction_value   AS november_transaction_value,
        total_transaction_value      AS december_transaction_value,
        transaction_value_mom_change AS transaction_value_change,
        transaction_count_mom_change_pct,
        transaction_value_mom_change_pct
    FROM intelligence.vw_ananse_customer_transaction_analysis
    WHERE year_month = '2025-12'
    ORDER BY transaction_value_mom_change DESC
    ),
customer_month_signals
AS (
    SELECT
        t50.ocb_customer_id,

recurrence_score
AS (
    SELECT
        ocb_customer_id,
        next_sequence_loan_id,
        next_sequence_loan_timestamp,
        recurrence_gap_minutes,
        loan_cashout_recurrence_score
    FROM intelligence.vw_loan_cashout_recurrence_score
    WHERE CONVERT(CHAR(7), next_sequence_loan_timestamp, 120) = '2025-11'
    ),
loan_cashout_signals
AS (
    SELECT
        ar.ocb_customer_id,
        ar.loan_id,
        ar.disbursement_timestamp,
        ar.transaction_id,
        ar.cash_out_amt,
        ar.cash_out_to_loan_ratio,
        ar.amount_ratio_score,
        ra.cash_out_delay_minutes,
        ra.loan_cashout_rapidity_score,
        re.recurrence_gap_minutes,
        re.loan_cashout_recurrence_score
    FROM amount_ratio_score ar
    LEFT JOIN rapidity_score ra ON
            ar.ocb_customer_id = ra.ocb_customer_id
                AND ar.loan_id = ra.loan_id
                AND ar.disbursement_timestamp = ra.disbursement_timestamp
    LEFT JOIN recurrence_score re ON
            ar.ocb_customer_id = re.ocb_customer_id
                AND ar.loan_id = re.next_sequence_loan_id
                AND ar.disbursement_timestamp = re.next_sequence_loan_timestamp
    INNER JOIN gold.vw_transaction t ON
            ar.transaction_id = t.transaction_id
                AND t.transaction_status = 'successful'
    )
SELECT
    t50.ocb_customer_id,
    t50.principal_disbursed,
    dr.reborrowing_rapidity_flag,
    dr.recycling_recurrence_flag,
    dr.repayment_deficiency_flag,
    COUNT(CASE 
            WHEN lc.amount_ratio_score > 0
                THEN 1
            END) AS elevated_amount_ratio_signal_count,
    COUNT(CASE 
            WHEN lc.loan_cashout_rapidity_score > 0
                THEN 1
            END) AS elevated_rapidity_signal_count,
    COUNT(CASE 
            WHEN lc.loan_cashout_recurrence_score > 0
                THEN 1
            END) AS elevated_recurrence_signal_count,
    COALESCE(dr.reborrowing_rapidity_flag, 0) + COALESCE(dr.recycling_recurrence_flag, 0) + COALESCE(dr.repayment_deficiency_flag, 0) + COUNT(CASE 
            WHEN lc.amount_ratio_score > 0
                THEN 1
            END) + COUNT(CASE 
            WHEN lc.loan_cashout_rapidity_score > 0
                THEN 1
            END) + COUNT(CASE 
            WHEN lc.loan_cashout_recurrence_score > 0
                THEN 1
            END) AS total_signal_count
FROM top_50 t50
LEFT JOIN loan_cashout_signals lc ON
        t50.ocb_customer_id = lc.ocb_customer_id
            AND t50.year_month = CONVERT(CHAR(7), lc.disbursement_timestamp, 120)
LEFT JOIN intelligence.vw_sikacredit_debt_recycling_customer_signals dr ON
        t50.ocb_customer_id = dr.ocb_customer_id
            AND t50.year_month = dr.year_month
GROUP BY t50.year_month,
    t50.ocb_customer_id,
    t50.principal_disbursed,
    t50.principal_disbursed_count,
    dr.reborrowing_rapidity_flag,
    dr.recycling_recurrence_flag,
    dr.repayment_deficiency_flag
ORDER BY t50.principal_disbursed DESC;

Some borrowers in SikaCredit's November 2025 lending-surge cohort exhibit both debt-recycling indicators and elevated
loan-disbursement-to-successful-cash-out signals. The overlap provides a basis for examining credit behaviour alongside subsequent
transaction behaviour, while preserving the distinction between the two analytical questions.

-- Oman Remit Institutional Trigger: Q4 2025 Remittance Activity
WITH quarterly_inbound AS
(
    SELECT
        DATEPART(YEAR, remittance_timestamp) AS reporting_year,
        DATEPART(QUARTER, remittance_timestamp) AS reporting_quarter,
        COUNT(*) AS inbound_remittance_count,
        COUNT(DISTINCT ocb_customer_id) AS distinct_customers,
        SUM(remittance_amount) AS inbound_remittance_value
    FROM gold.vw_remittance
    WHERE remittance_type = 'remittance_received'
      AND remittance_status = 'successful'
    GROUP BY
        DATEPART(YEAR, remittance_timestamp),
        DATEPART(QUARTER, remittance_timestamp)
),
quarterly_changes AS
(
    SELECT
        reporting_year,
        reporting_quarter,
        inbound_remittance_count,
        distinct_customers,
        inbound_remittance_value,

        LAG(inbound_remittance_count) OVER (
            ORDER BY reporting_year, reporting_quarter
        ) AS previous_quarter_count,

        LAG(distinct_customers) OVER (
            ORDER BY reporting_year, reporting_quarter
        ) AS previous_quarter_customers,

        LAG(inbound_remittance_value) OVER (
            ORDER BY reporting_year, reporting_quarter
        ) AS previous_quarter_value
    FROM quarterly_inbound
)
SELECT
    CONCAT(reporting_year, '-Q', reporting_quarter) AS year_quarter,
    inbound_remittance_count,
    distinct_customers,
    inbound_remittance_value,

    inbound_remittance_count - previous_quarter_count
        AS count_qoq_change,

    CAST(
        100.0 * (inbound_remittance_count - previous_quarter_count)
        / NULLIF(previous_quarter_count, 0)
        AS DECIMAL(10, 2)
    ) AS count_qoq_pct,

    distinct_customers - previous_quarter_customers
        AS customers_qoq_change,

    CAST(
        100.0 * (distinct_customers - previous_quarter_customers)
        / NULLIF(previous_quarter_customers, 0)
        AS DECIMAL(10, 2)
    ) AS customers_qoq_pct,

    inbound_remittance_value - previous_quarter_value
        AS value_qoq_change,

    CAST(
        100.0 * (inbound_remittance_value - previous_quarter_value)
        / NULLIF(previous_quarter_value, 0)
        AS DECIMAL(10, 2)
    ) AS value_qoq_pct
FROM quarterly_changes
WHERE CONCAT(reporting_year, '-Q', reporting_quarter) IN ('2025-Q3', '2025-Q4')
ORDER BY reporting_year, reporting_quarter;

LEFT JOIN intelligence.vw_credit_to_remittance_recurrence_score AS rc
    ON rc.next_remittance_id = q.remittance_id
   AND rc.ocb_customer_id = q.ocb_customer_id

WHERE rp.loan_id IS NOT NULL
)
SELECT
    ocb_customer_id,

    COUNT(*) AS qualifying_sequence_count,

    SUM(CASE
            WHEN credit_to_remittance_rapidity_score > 0 THEN 1
            ELSE 0
        END) AS positive_rapidity_sequence_count,

    MAX(credit_to_remittance_rapidity_score) AS max_rapidity_score,

    SUM(CASE
            WHEN credit_to_remittance_value_correspondence_score > 0 THEN 1
            ELSE 0
        END) AS positive_value_correspondence_sequence_count,

    MAX(credit_to_remittance_value_correspondence_score)
        AS max_value_correspondence_score,

    SUM(CASE
            WHEN credit_to_remittance_recurrence_score > 0 THEN 1
            ELSE 0
        END) AS positive_recurrence_sequence_count,

    MAX(credit_to_remittance_recurrence_score) AS max_recurrence_score

FROM qualifying_sequences
GROUP BY ocb_customer_id
ORDER BY ocb_customer_id;










ORDER BY
    q.ocb_customer_id,
    q.remittance_timestamp,
    q.remittance_id;
